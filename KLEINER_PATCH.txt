  [107,'Becky Bat',6,8,3,6]

  [107,'Becky Bat',6,8,3,6],

  n>=94&&n<=107

  n>=94&&n<=108

  [1,43,44,51,52,53,56,58,94,95,96,97,98,99,100,101,102,103,104,105,106,107].includes(n)
